# Запуск: node app.js (3000), node example.js (3001).
# PowerShell: использовать curl.exe; Linux/macOS: curl.
# Выполнять последовательно после свежего запуска серверов.

# Практический пример: список
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3001/items"

# Практический пример: один товар
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3001/items/2"

# Практический пример: создание
curl.exe --silent --show-error --include --max-time 10 --request POST "http://127.0.0.1:3001/items" --header "Content-Type: application/json" --data-binary "@checks/example-create-request.json"

# Практический пример: обновление
curl.exe --silent --show-error --include --max-time 10 --request PUT "http://127.0.0.1:3001/items/1" --header "Content-Type: application/json" --data-binary "@checks/example-update-request.json"

# Практический пример: удаление
curl.exe --silent --show-error --include --max-time 10 --request DELETE "http://127.0.0.1:3001/items/3"

# Автомобили: список
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars"

# Автомобили: один автомобиль
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars/2"

# Автомобили: создание
curl.exe --silent --show-error --include --max-time 10 --request POST "http://127.0.0.1:3000/cars" --header "Content-Type: application/json" --data-binary "@checks/cars-create-request.json"

# Автомобили: чтение созданного
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars/6"

# Автомобили: полная замена
curl.exe --silent --show-error --include --max-time 10 --request PUT "http://127.0.0.1:3000/cars/6" --header "Content-Type: application/json" --data-binary "@checks/cars-update-request.json"

# Автомобили: поиск по марке
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?search=tOyOtA"

# Автомобили: цена по возрастанию
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?sort=price&order=asc"

# Автомобили: цена по убыванию
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?sort=price&order=desc"

# Автомобили: страница 2
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?page=2&limit=2"

# Автомобили: поиск + сортировка + страница
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?search=toyota&sort=price&order=desc&page=1&limit=1"

# Ошибка: отсутствуют обязательные поля
curl.exe --silent --show-error --include --max-time 10 --request POST "http://127.0.0.1:3000/cars" --header "Content-Type: application/json" --data-binary "@checks/cars-missing-request.json"

# Ошибка: price передан строкой
curl.exe --silent --show-error --include --max-time 10 --request POST "http://127.0.0.1:3000/cars" --header "Content-Type: application/json" --data-binary "@checks/cars-type-request.json"

# Ошибка: недопустимые значения
curl.exe --silent --show-error --include --max-time 10 --request POST "http://127.0.0.1:3000/cars" --header "Content-Type: application/json" --data-binary "@checks/cars-values-request.json"

# Ошибка: PUT требует все поля
curl.exe --silent --show-error --include --max-time 10 --request PUT "http://127.0.0.1:3000/cars/6" --header "Content-Type: application/json" --data-binary "@checks/cars-put-missing-request.json"

# Ошибка: неверные query-параметры
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars?page=0&limit=2"

# Ошибка: автомобиль не найден
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars/999"

# Ошибка: обновление отсутствующего
curl.exe --silent --show-error --include --max-time 10 --request PUT "http://127.0.0.1:3000/cars/999" --header "Content-Type: application/json" --data-binary "@checks/cars-put-not-found-request.json"

# Ошибка: маршрут не найден
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/unknown"

# Автомобили: удаление без тела ответа
curl.exe --silent --show-error --include --max-time 10 --request DELETE "http://127.0.0.1:3000/cars/6"

# Проверка: удалённый автомобиль не найден
curl.exe --silent --show-error --include --max-time 10 --request GET "http://127.0.0.1:3000/cars/6"

# Проверка: повторное удаление
curl.exe --silent --show-error --include --max-time 10 --request DELETE "http://127.0.0.1:3000/cars/6"
